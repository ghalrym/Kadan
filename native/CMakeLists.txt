cmake_minimum_required(VERSION 3.24)
project(kadan_native LANGUAGES CXX)
option(KADAN_ENABLE_CUDA "Build original SM86 projection and unexecuted GPU parity harness" OFF)
if(KADAN_ENABLE_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86)
  endif()
  if(NOT "${CMAKE_CUDA_ARCHITECTURES}" STREQUAL "86")
    message(FATAL_ERROR "The first native GPU path requires explicit SM86 (86), never native device probing")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  # Compiler checks only. No CUDA device test is registered in CTest.
  add_library(kadan_cuda INTERFACE)
  target_link_libraries(kadan_cuda INTERFACE CUDA::cudart)
  target_compile_features(kadan_cuda INTERFACE cuda_std_20)
endif()
find_package(Threads REQUIRED)
add_library(kadan_resources INTERFACE)
target_include_directories(kadan_resources INTERFACE include)
target_compile_features(kadan_resources INTERFACE cxx_std_20)
target_link_libraries(kadan_resources INTERFACE Threads::Threads)
add_executable(kadan-worker src/worker.cpp)
target_link_libraries(kadan-worker PRIVATE kadan_resources)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kadan-worker PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(kadan_quantization src/quantization.cpp)
target_include_directories(kadan_quantization PUBLIC include)
target_compile_features(kadan_quantization PUBLIC cxx_std_20)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kadan_quantization PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(kadan_checkpoint src/checkpoint.cpp)
target_include_directories(kadan_checkpoint PUBLIC include)
target_compile_features(kadan_checkpoint PUBLIC cxx_std_20)
target_link_libraries(kadan_checkpoint PUBLIC kadan_quantization Threads::Threads)
add_executable(kadan-checkpoint-inspect src/checkpoint_inspect.cpp)
target_link_libraries(kadan-checkpoint-inspect PRIVATE kadan_checkpoint)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kadan_checkpoint PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_compile_options(kadan-checkpoint-inspect PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_library(kadan_cuda_plan src/cuda_plan.cpp)
target_include_directories(kadan_cuda_plan PUBLIC include)
target_compile_features(kadan_cuda_plan PUBLIC cxx_std_20)
target_link_libraries(kadan_cuda_plan PUBLIC kadan_quantization)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(kadan_cuda_plan PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(KADAN_ENABLE_CUDA)
  add_library(kadan_gpu src/cuda_projection.cpp src/nvfp4_kernel.cu)
  target_link_libraries(kadan_gpu PUBLIC kadan_cuda_plan kadan_resources kadan_cuda)
  set_target_properties(kadan_gpu PROPERTIES CUDA_STANDARD 20 CUDA_STANDARD_REQUIRED YES CUDA_ARCHITECTURES 86)
  target_compile_options(kadan_gpu PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--default-stream=legacy;--ftz=false>)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(kadan_gpu PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-Wall;-Wextra;-Wpedantic;-Werror>)
  endif()
  add_executable(kadan-cuda-parity tests/cuda_parity.cpp)
  target_link_libraries(kadan-cuda-parity PRIVATE kadan_gpu)
  # Deliberately not added to CTest: execution requires explicit GPU authorization.
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(resource-tests tests/resources.cpp)
  target_link_libraries(resource-tests PRIVATE kadan_resources)
  add_test(NAME resources COMMAND resource-tests)
  add_test(NAME worker-protocol COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/protocol.py" "$<TARGET_FILE:kadan-worker>")
  add_executable(quantization-tests tests/quantization.cpp)
  target_link_libraries(quantization-tests PRIVATE kadan_quantization)
  add_test(NAME quantization COMMAND quantization-tests)
  add_executable(checkpoint-tests tests/checkpoint.cpp)
  target_link_libraries(checkpoint-tests PRIVATE kadan_checkpoint)
  add_test(NAME checkpoint COMMAND checkpoint-tests)
  add_test(NAME checkpoint-inspect-usage COMMAND kadan-checkpoint-inspect)
  set_tests_properties(checkpoint-inspect-usage PROPERTIES WILL_FAIL TRUE)
  add_executable(cuda-plan-tests tests/cuda_plan.cpp)
  target_link_libraries(cuda-plan-tests PRIVATE kadan_cuda_plan)
  add_test(NAME cuda-plan COMMAND cuda-plan-tests)
  set_tests_properties(resources worker-protocol quantization checkpoint checkpoint-inspect-usage cuda-plan PROPERTIES TIMEOUT 15)
endif()
